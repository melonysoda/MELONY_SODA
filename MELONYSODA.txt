    // 투표 기능
    function toggleVote(id, type) {
        const place = places.find(p => p.id === id);
        if(!place) return;
        const currentUser = prompt("투표자의 이름을 입력해 주세요 (" + members.join(', ') + ")");
        if(!currentUser || !members.includes(currentUser)) {
            if(currentUser) alert("올바른 멤버 이름을 입력해주세요.");
            return;
        }

        let upvotes = place.upvotes || [];
        let downvotes = place.downvotes || [];

        if(type === 'up') {
            if(upvotes.includes(currentUser)) {
                upvotes = upvotes.filter(u => u !== currentUser);
            } else {
                upvotes.push(currentUser);
                downvotes = downvotes.filter(u => u !== currentUser);
            }
        } else {
            if(downvotes.includes(currentUser)) {
                downvotes = downvotes.filter(u => u !== currentUser);
            } else {
                downvotes.push(currentUser);
                upvotes = upvotes.filter(u => u !== currentUser);
            }
        }

        updateDoc(doc(db, "japan_trip_db", id), { upvotes, downvotes });
    }

        const min = document.getElementById('custom-sched-min').value;
        const title = document.getElementById('custom-sched-title').value.trim();
        const memo = document.getElementById('custom-sched-memo').value.trim();
        const selectedMembers = Array.from(document.querySelectorAll('.custom-member-cb:checked')).map(cb => cb.value);

        await addDoc(collection(db, "japan_schedules"), {
            day,
            time: `${hour}:${min}`,
            title,
            url: '',
            memo,
            members: selectedMembers,
            createdAt: Date.now()
        });

        document.getElementById('custom-sched-title').value = '';
        document.getElementById('custom-sched-memo').value = '';
    });

    // 타임라인 렌더링
    function renderTimeline() {
        const container = document.getElementById('timeline-container');
        if(!container) return;

        let html = "";
        tripDays.forEach(day => {
            const dayScheds = schedules.filter(s => s.day === day);
            dayScheds.sort((a, b) => a.time.localeCompare(b.time));

            html += `<div class="timeline-day-header">📅 Day ${day}</div>`;

            if(dayScheds.length === 0) {
                html += `<div class="empty-msg" style="padding: 15px 0;">등록된 일정이 없습니다.</div>`;
            } else {
                dayScheds.forEach(s => {
                    const isEditing = editingSchedId === s.id;
                    if(isEditing) {
                        html += `<div class="timeline-item">
                            <div class="timeline-card" style="width:100%;">
                                <div class="edit-mode-box">
                                    <input type="text" id="edit-sched-title-${s.id}" value="${s.title}">
                                    <input type="text" id="edit-sched-memo-${s.id}" value="${s.memo || ''}">
                                    <div style="display:flex; gap:6px;">
                                        <button class="save-btn" onclick="saveSchedEdit('${s.id}')">💾 저장</button>
                                        <button class="edit-btn" onclick="cancelSchedEdit()">취소</button>
                                    </div>
                                </div>
                            </div>
                        </div>`;
                    } else {
                        html += `<div class="timeline-item">
                            <div class="timeline-time">${s.time}</div>
                            <div class="timeline-card">
                                <div class="sched-title">${s.url ? `<a href="${s.url}" target="_blank">🔗 ${s.title}</a>` : s.title}</div>
                                ${s.memo ? `<div class="sched-memo">${s.memo}</div>` : ''}
                                ${s.members && s.members.length > 0 ? `<div style="font-size:11px; color:var(--text-muted); margin-top:6px;">👥 참여: ${s.members.join(', ')}</div>` : ''}
                                <div class="sched-actions">
                                    <button class="edit-btn" onclick="turnOnSchedEditMode('${s.id}')">✏️</button>
                                    <button class="delete-btn" onclick="deleteSched('${s.id}')">🗑️</button>
                                </div>
                            </div>
                        </div>`;
                    }
                });
            }
        });

        container.innerHTML = html;
    }

    function deleteSched(id) {
        if(confirm("이 일정을 삭제하시겠습니까?")) {
            deleteDoc(doc(db, "japan_schedules", id));
        }
    }

    function turnOnSchedEditMode(id) {
        editingSchedId = id;
        renderTimeline();
    }

    function cancelSchedEdit() {
        editingSchedId = null;
        renderTimeline();
    }

    function saveSchedEdit(id) {
        const title = document.getElementById(`edit-sched-title-${id}`).value.trim();
        const memo = document.getElementById(`edit-sched-memo-${id}`).value.trim();

        if(!title) { alert("일정 제목을 입력해 주세요."); return; }

        updateDoc(doc(db, "japan_schedules", id), { title, memo });
        editingSchedId = null;
    }
